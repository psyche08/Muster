cmake_minimum_required(VERSION 3.24)
project(muster_remote_desktop_checks LANGUAGES C CXX)

set(MOONLIGHT_CORE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/moonlight-pc/moonlight-common-c/moonlight-common-c")
add_subdirectory("${MOONLIGHT_CORE}" moonlight-common-c)

if(WIN32 AND NOT USE_MBEDTLS)
    target_link_libraries(moonlight-common-c PRIVATE crypt32)
endif()

enable_testing()
add_executable(moonlight_loopback_test tests/moonlight_loopback.c)
target_include_directories(moonlight_loopback_test PRIVATE "${MOONLIGHT_CORE}/src")
target_link_libraries(moonlight_loopback_test PRIVATE moonlight-common-c)
add_test(NAME moonlight_loopback COMMAND moonlight_loopback_test)

add_executable(sunshine_bootstrap_test tests/sunshine_bootstrap.cpp)
target_compile_features(sunshine_bootstrap_test PRIVATE cxx_std_20)
add_test(NAME sunshine_bootstrap COMMAND sunshine_bootstrap_test)

option(MUSTER_BUILD_WINDOWS_VIEWER "Build the separate native Windows viewer" OFF)
if(MUSTER_BUILD_WINDOWS_VIEWER)
    if(NOT WIN32)
        message(FATAL_ERROR "The native Windows viewer requires a Windows target")
    endif()
    find_package(FFMPEG REQUIRED)
    add_library(muster-viewer-video STATIC viewer/video.cpp)
    add_executable(muster-moonlight WIN32 viewer/windows.cpp)
    add_executable(windows_viewer_test tests/windows_viewer.cpp)
    add_test(NAME windows_viewer COMMAND windows_viewer_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/blue-720p.h264")
    target_compile_features(muster-viewer-video PUBLIC cxx_std_20)
    target_compile_definitions(muster-viewer-video PUBLIC NOMINMAX UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
    target_include_directories(muster-viewer-video PRIVATE ${FFMPEG_INCLUDE_DIRS})
    target_link_directories(muster-viewer-video PUBLIC ${FFMPEG_LIBRARY_DIRS})
    target_link_libraries(muster-moonlight PRIVATE muster-viewer-video)
    target_link_libraries(windows_viewer_test PRIVATE muster-viewer-video)
    target_link_libraries(muster-viewer-video PUBLIC moonlight-common-c ${FFMPEG_LIBRARIES} user32 gdi32)
endif()

if(APPLE)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        add_library(muster-moonlight STATIC viewer/apple.c)
    else()
        add_executable(muster-moonlight viewer/apple.c)
        target_compile_definitions(muster-moonlight PRIVATE MUSTER_MOONLIGHT_CHILD)
    endif()
    target_link_libraries(muster-moonlight PRIVATE moonlight-common-c)
endif()
